\fsection[\textcolor{waveBlue2}{Ejercicio 11}]{\boldit{\textcolor{waveBlue2}{TU SECTOR PROFESIONAL}}{
Indica qué principios del ecodiseño no cumplen los siguientes productos y explica qué cambios deberían realizarse para que tengan en cuenta aspectos ambientales:
}}

\begin{solution}
\begin{enumerate}[label=\alph*)]
    \item \textbf{Camiseta fabricada en la India}
    \begin{itemize}
        \item \textbf{Principios de ecodiseño no cumplidos:}
            \begin{itemize}
                \item Uso de materiales de baja calidad → baja durabilidad.
                \item Uso de materiales plásticos y productos químicos → no reciclable ni biodegradable.
                \item Producción localizada lejos del consumidor → alta huella de transporte.
            \end{itemize}
        \item \textbf{Cambios para mejorar sostenibilidad:}
            \begin{itemize}
                \item Usar materiales de alta calidad, naturales o reciclados.
                \item Sustituir productos químicos por alternativas menos contaminantes.
                \item Producir localmente para reducir transporte y emisiones.
            \end{itemize}
    \end{itemize}

    \item \textbf{Frutas procedentes de Sudamérica}
    \begin{itemize}
        \item \textbf{Principios de ecodiseño no cumplidos:}
            \begin{itemize}
                \item Agricultura intensiva → alto uso de plaguicidas y fertilizantes químicos.
                \item Transporte en avión → gran huella de carbono.
                \item Envasado plástico → problemas de reciclabilidad y residuos.
            \end{itemize}
        \item \textbf{Cambios para mejorar sostenibilidad:}
            \begin{itemize}
                \item Fomentar agricultura ecológica o integrada, reduciendo productos químicos.
                \item Transporte por medios más sostenibles (barco o producción local si es posible).
                \item Uso de envases reciclables, compostables o eliminación de envases innecesarios.
            \end{itemize}
    \end{itemize}
\end{enumerate}
\end{solution}
