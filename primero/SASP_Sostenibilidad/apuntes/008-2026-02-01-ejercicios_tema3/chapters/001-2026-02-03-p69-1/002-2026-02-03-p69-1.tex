\fsection{\boldit{\textcolor{waveBlue2}{Actividades:}}} 
\subsection*{Una organización empresarial al implantar un plan de cultura de sostenibilidad, pone en marcha una serie de mecanismos y resortes que afectan a los distintos espacios en la búsqueda de diferentes efectos.
En la consecución de los beneficios que a continuación se enumeran, realiza una breve exposición de medidas que se podrían implementar para que se produjeran los efectos deseados:}

\begin{solution}
	\begin{enumerate}[label=\alph*)]
		\item \textbf{Un control del consumo energético mas eficiente.}

			\begin{itemize}

				\item \textbf{Instalación de sensores de movimiento e iluminación LED automática en almacenes y oficinas.}
				\item \textbf{Implementación de sistemas de climatización inteligente con termostatos programables.}
				\item \textbf{Auditorías energéticas periódicas para identificar y corregir puntos de alto consumo.}
				
			\end{itemize}


		\item \textbf{Utilización de medios de transporte mas sostenibles.}

			\begin{itemize}

				\item \textbf{Renovación de la flota de reparto con vehículos eléctricos o de bajas emisiones.}
				\item \textbf{Optimización de rutas de distribución mediante software logístico para reducir kilómetros recorridos.}
				\item \textbf{Fomento del transporte combinado (ferrocarril + carretera) para larga distancia.}
				
			\end{itemize}


		\item \textbf{Una gestión optima de los recursos generados por la emrpresa.}
			
			\begin{itemize}

				\item \textbf{Implementación de un sistema de economía circular: reutilización de embalajes, palés y materiales.}
				\item \textbf{Programa de separación y reciclaje de residuos orgánicos, plásticos, cartón y otros materiales.}
				\item \textbf{Donación sistemática de excedentes alimentarios aptos para el consumo a bancos de alimentos.}
				
			\end{itemize}


		\item \textbf{Colaborar activamente con empresas locales y llevar a cabo acciones sociales.}

			\begin{itemize}

				\item \textbf{Creación de un programa de compra local preferente para productos de proximidad (km 0).}
				\item \textbf{Colaboración con proveedores locales en campañas de marketing conjunto.}
				\item \textbf{Desarrollo de programas de voluntariado corporativo en colaboración con ONGs locales.}
				
			\end{itemize}



	\end{enumerate}
\end{solution}
