\fsubsection
[\textcolor{waveBlue2}{Tarea: Registro de Actividades de Tratamiento (RAT)}]
{\textcolor{waveBlue2}{\boldit{Tarea: Registro de Actividades de Tratamiento (RAT)}}}
{
	El RGPD exige a los responsables de los datos de una empresa que mantengan los registros de
	actividades de tratamiento de datos que tienen bajo su responsabilidad y control. Elige un
	negocio relacionado con la familia profesional del ciclo formativo que estás cursando y realiza
	un Registro de Actividades de Tratamiento (RAT) usando la herramienta Facilita RGPD
	(\textless\,\url{https://bit.ly/3T0zsiO}\,\textgreater).
}

\begin{solution}

	\textbf{Empresa ficticia:} \textit{AutoSistemas Industriales S.L.} \\
	Empresa de mantenimiento e instalación de sistemas de automatización industrial.

	\medskip

	\textbf{Registro de Actividades de Tratamiento (RAT) -- Art. 30 RGPD}

	\medskip

	\begin{enumerate}[label=\textbf{\arabic*.}]

		\item \textbf{Identificación del responsable del tratamiento}
		      \begin{itemize}
			      \item Nombre: AutoSistemas Industriales S.L.
			      \item CIF: B-XXXXXXXX
			      \item Dirección: Polígono Industrial Las Eras, Cuenca
			      \item Correo electrónico: info@autosistemas.es
			      \item Delegado de Protección de Datos (DPD): No aplica (menos de 250 empleados,
			            tratamiento no sistemático de datos sensibles).
		      \end{itemize}

		\item \textbf{Actividad de tratamiento 1: Gestión de clientes}
		      \begin{itemize}
			      \item \textbf{Finalidad:} Gestión de contratos de mantenimiento y facturación a clientes.
			      \item \textbf{Categorías de interesados:} Clientes personas físicas y representantes
			            legales de empresas clientes.
			      \item \textbf{Categorías de datos:} Datos identificativos (nombre, NIF/CIF, dirección),
			            datos de contacto (teléfono, correo electrónico), datos económicos (datos bancarios
			            para domiciliación de pagos).
			      \item \textbf{Destinatarios:} Gestoría externa (encargada del tratamiento), entidades
			            bancarias.
			      \item \textbf{Transferencias internacionales:} No se realizan.
			      \item \textbf{Plazo de supresión:} 5 años desde la finalización del contrato (obligación
			            fiscal y mercantil).
			      \item \textbf{Base jurídica:} Ejecución de contrato (art. 6.1.b RGPD).
		      \end{itemize}

		\item \textbf{Actividad de tratamiento 2: Gestión de empleados}
		      \begin{itemize}
			      \item \textbf{Finalidad:} Gestión de nóminas, altas en Seguridad Social y prevención
			            de riesgos laborales.
			      \item \textbf{Categorías de interesados:} Empleados de la empresa.
			      \item \textbf{Categorías de datos:} Datos identificativos (nombre, DNI, número de
			            afiliación SS), datos económicos (salario, cuenta bancaria), datos de salud (partes
			            de baja médica -- categoría especial).
			      \item \textbf{Destinatarios:} Mutua laboral, Seguridad Social, Agencia Tributaria,
			            gestoría de nóminas.
			      \item \textbf{Transferencias internacionales:} No se realizan.
			      \item \textbf{Plazo de supresión:} 4 años (obligaciones tributarias) / 10 años
			            (prevención de riesgos, art. 23 LPRL).
			      \item \textbf{Base jurídica:} Cumplimiento de obligación legal (art. 6.1.c RGPD);
			            datos de salud: art. 9.2.b RGPD.
		      \end{itemize}
	\end{enumerate}
\end{solution}

		\item \textbf{Actividad de tratamiento 3: Gestión de proveedores}
		      \begin{itemize}
			      \item \textbf{Finalidad:} Gestión de pedidos de componentes y equipos industriales,
			            así como pago a proveedores.
			      \item \textbf{Categorías de interesados:} Representantes y contactos de empresas
			            proveedoras.
			      \item \textbf{Categorías de datos:} Datos identificativos y de contacto profesional,
			            datos económicos (IBAN, condiciones comerciales).
			      \item \textbf{Destinatarios:} Entidades bancarias, gestoría.
			      \item \textbf{Transferencias internacionales:} No se realizan.
			      \item \textbf{Plazo de supresión:} 5 años desde la última operación.
			      \item \textbf{Base jurídica:} Ejecución de contrato / interés legítimo (art. 6.1.b
			            y 6.1.f RGPD).
		      \end{itemize}

	\end{enumerate}
