\fsection[\textcolor{waveBlue2}{Ejercicio 2}]{\boldit{\textcolor{waveBlue2}{Actividades}}{
Indica al cumplimiento de qué metas del ODS 12 contribuyen las siguientes acciones :
}}

\begin{solution}


    \begin{enumerate}[label=\alph*)]
    \item \textbf{Planificar el menú y aprovechar sobras}  
    \begin{itemize}
        \item Meta 12.3: Reducir el desperdicio de alimentos a nivel de consumidores.
    \end{itemize}

    \item \textbf{Sustituir envases de plástico por materiales reciclables}  
    \begin{itemize}
        \item Meta 12.5: Reducir la generación de residuos mediante reciclaje y reutilización.
    \end{itemize}

    \item \textbf{Talleres de reciclaje en colegios}  
    \begin{itemize}
        \item Meta 12.8: Garantizar información y educación para el desarrollo sostenible.
    \end{itemize}

    \item \textbf{Eliminar subvenciones al carbón}  
    \begin{itemize}
        \item Meta 12.c: Racionalizar y eliminar subsidios ineficientes a los combustibles fósiles.
    \end{itemize}

    \item \textbf{Aplicación Too Good To Go (venta de excedentes de comida)}  
    \begin{itemize}
        \item Meta 12.3: Reducir el desperdicio alimentario en comercio y consumo.
    \end{itemize}
\end{enumerate}
\end{solution}
